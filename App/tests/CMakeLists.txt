# w2-061: unit tests of App/appconfig.{h,cpp} (config.json reader, desktop) and
# App/runtimeinfo.{h,cpp} (/runtime.json format + request). Separate small CTest project so
# the application's Qt Design Studio CMake is not touched; it compiles the application's own
# source files (same folder tree, nothing copied). See README.md.
cmake_minimum_required(VERSION 3.22)
project(TaidaFlowAppTests LANGUAGES CXX)

set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Test)
qt_standard_project_setup()
enable_testing()

set(APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
# Qt DLLs for ctest runs (no PATH setup needed).
set(QT_BIN_DIR "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_BINS}")

function(taidaflow_add_test name)
    qt_add_executable(${name} ${ARGN})
    set_target_properties(${name} PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE FALSE)
    target_include_directories(${name} PRIVATE "${APP_DIR}")
    target_link_libraries(${name} PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    # Plain-text result per test in the build folder (ctest does not always echo QTest's
    # stdout on Windows) + the same on stdout.
    add_test(NAME ${name} COMMAND ${name} -o "${CMAKE_CURRENT_BINARY_DIR}/${name}.result.txt,txt" -o "-,txt")
    set_tests_properties(${name} PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${QT_BIN_DIR}"
        TIMEOUT 120)
endfunction()

taidaflow_add_test(tst_appconfig
    tst_appconfig.cpp
    "${APP_DIR}/appconfig.cpp"
    "${APP_DIR}/appconfig.h")

taidaflow_add_test(tst_runtimeinfo
    tst_runtimeinfo.cpp
    "${APP_DIR}/runtimeinfo.cpp"
    "${APP_DIR}/runtimeinfo.h")

# w2-064: App/applog.{h,cpp} (the app's own daily log files, spec §2 "log").
taidaflow_add_test(tst_applog

    "${APP_DIR}/applog.cpp"
    "${APP_DIR}/applog.h"

# ---------------------------------------------------------------------------------------------
# w2-082: interface font of the core branch = the embedded Noto Sans TC subset, desktop and web.
# The code under test is NOT copied: at configure time the exact text of App/main.cpp is cut into
# three generated snippets (main.cpp is a configure dependency, so any edit re-generates them):
#   uifont_applyuifont.inc      QString applyUiFont(...) { ... }          (the whole function)
#   uifont_sequence_*.inc       from `const QString cjkFamily = ...` (the "(core only)" block)
#                               to the end of the Proxy-free part, plus the HOOK block:
#                               `const QString uiFontFamilyRequest = ...; applyUiFont(...);`
# tst_uifont runs the sequence as compiled for the desktop; tst_uifont_wasmpath runs the same text
# with `#if defined(Q_OS_WASM)` switched on (installCjkFallbackChain + applyUiFont, the web order)
# on the desktop font database. Both use App/embeddedfonts.cpp and the committed App/fonts/*.ttf.
# ---------------------------------------------------------------------------------------------
find_package(Qt6 6.8 REQUIRED COMPONENTS Gui Widgets Qml Quick)
set(UIFONT_MAIN_CPP "${APP_DIR}/main.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${UIFONT_MAIN_CPP}")
file(READ "${UIFONT_MAIN_CPP}" _uifont_main)
string(REPLACE "\r\n" "\n" _uifont_main "${_uifont_main}")

# Cuts [begin marker, end marker) out of `text` (end searched after begin); FATAL_ERROR when missing.
function(uifont_cut text begin_marker end_marker out_var)
    string(FIND "${text}" "${begin_marker}" _b)
    if(_b LESS 0)
        message(FATAL_ERROR "[w2-082] App/main.cpp: marker not found: ${begin_marker}")
    endif()
    string(SUBSTRING "${text}" ${_b} -1 _rest)
    string(FIND "${_rest}" "${end_marker}" _e)
    if(_e LESS 0)
        message(FATAL_ERROR "[w2-082] App/main.cpp: end marker not found after '${begin_marker}': ${end_marker}")
    endif()
    string(SUBSTRING "${_rest}" 0 ${_e} _cut)
    set(${out_var} "${_cut}" PARENT_SCOPE)
endfunction()

uifont_cut("${_uifont_main}" "QString applyUiFont(const QString &requested)"
           "// ===== end of UI font" _uifont_fn)
uifont_cut("${_uifont_main}" "const QString cjkFamily = TaidaFlowFonts::loadEmbeddedCjkFont();"
           "// ===== Proxy composition" _uifont_coreonly)
# Line start + indentation: the HOOK comment above the line quotes the same statement.
uifont_cut("${_uifont_main}" "\n    const QString uiFontFamilyRequest"
           "// ===== end of UI font" _uifont_hook)
string(FIND "${_uifont_coreonly}" "#if defined(Q_OS_WASM)" _uifont_wasm_if)
if(_uifont_wasm_if LESS 0)
    message(FATAL_ERROR "[w2-082] App/main.cpp: the (core only) block has no '#if defined(Q_OS_WASM)'")
endif()
string(REPLACE "#if defined(Q_OS_WASM)" "#if 1 // [test] Q_OS_WASM branch of main.cpp" _uifont_coreonly_wasm "${_uifont_coreonly}")
set(_uifont_gen "${CMAKE_CURRENT_BINARY_DIR}/uifont_gen")
set(_uifont_note "// Generated by App/tests/CMakeLists.txt (w2-082) from App/main.cpp - do not edit.\n")
file(CONFIGURE OUTPUT "${_uifont_gen}/uifont_applyuifont.inc" CONTENT "${_uifont_note}${_uifont_fn}" @ONLY)
file(CONFIGURE OUTPUT "${_uifont_gen}/uifont_sequence_desktop.inc" CONTENT "${_uifont_note}${_uifont_coreonly}${_uifont_hook}" @ONLY)
file(CONFIGURE OUTPUT "${_uifont_gen}/uifont_sequence_wasm.inc" CONTENT "${_uifont_note}${_uifont_coreonly_wasm}${_uifont_hook}" @ONLY)

function(taidaflow_add_uifont_test name wasm_path)
    taidaflow_add_test(${name}
        tst_uifont.cpp
        "${APP_DIR}/embeddedfonts.cpp"
        "${APP_DIR}/embeddedfonts.h")
    qt_add_resources(${name} "${name}_embedded_fonts"
        PREFIX "/fonts"
        BASE "${APP_DIR}/fonts"
        FILES
            "${APP_DIR}/fonts/TaidaFlowNotoSansTC-Regular.ttf"
            "${APP_DIR}/fonts/TaidaFlowNotoSansTC-Bold.ttf")
    target_include_directories(${name} PRIVATE "${_uifont_gen}")
    target_compile_definitions(${name} PRIVATE
        UIFONT_WASM_PATH=${wasm_path}
        UIFONT_MAIN_CPP="${UIFONT_MAIN_CPP}"
        UIFONT_APP_QML="${APP_DIR}/../TaidaFlowContent/App.qml"
        UIFONT_CHARSET="${APP_DIR}/fonts/charset.txt")
    target_link_libraries(${name} PRIVATE Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick)
endfunction()
taidaflow_add_uifont_test(tst_uifont 0)
taidaflow_add_uifont_test(tst_uifont_wasmpath 1)
